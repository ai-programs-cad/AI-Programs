cmake_minimum_required(VERSION 3.24)
project(CADPreview VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(CADPREVIEW_WITH_OCCT "STEP support (required for release)" ON)
add_library(CADPreview.Core src/core.cpp src/dxf.cpp src/stl.cpp src/scene_io.cpp)
target_include_directories(CADPreview.Core PUBLIC include)
if(CADPREVIEW_WITH_OCCT)
  if(CADPREVIEW_OCCT_BUILD_ROOT)
    set(OpenCASCADE_INCLUDE_DIR "${CADPREVIEW_OCCT_BUILD_ROOT}/inc")
    set(OpenCASCADE_LIBRARY_DIR "${CADPREVIEW_OCCT_BUILD_ROOT}/win64/clang/lib")
  else()
    find_package(OpenCASCADE 7.8 REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms ApplicationFramework DataExchange)
  endif()
  target_sources(CADPreview.Core PRIVATE src/step.cpp)
  target_include_directories(CADPreview.Core PRIVATE ${OpenCASCADE_INCLUDE_DIR})
  target_compile_definitions(CADPreview.Core PRIVATE CADPREVIEW_WITH_OCCT)
  target_link_directories(CADPreview.Core PUBLIC ${OpenCASCADE_LIBRARY_DIR})
  target_link_libraries(CADPreview.Core PUBLIC TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKMesh TKShHealing TKCDF TKLCAF TKCAF TKXCAF TKDE TKDESTEP TKXSBase)
endif()
enable_testing()
add_executable(CADPreview.Tests tests/tests.cpp)
target_link_libraries(CADPreview.Tests PRIVATE CADPreview.Core)
add_test(NAME geometry-and-corruption COMMAND CADPreview.Tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)
set_tests_properties(geometry-and-corruption PROPERTIES TIMEOUT 60)
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  add_library(CADPreview.Renderer src/renderer.cpp src/worker_client.cpp src/viewport.cpp src/settings.cpp src/help_text.cpp)
  target_include_directories(CADPreview.Renderer PUBLIC include)
  target_link_libraries(CADPreview.Renderer PUBLIC CADPreview.Core gdi32 user32 comctl32 comdlg32 shell32 ole32 advapi32)
  add_executable(CADPreview.Worker src/worker.cpp)
  target_link_libraries(CADPreview.Worker PRIVATE CADPreview.Renderer)
  add_executable(CADPreview.Viewer WIN32 src/viewer.cpp)
  target_link_libraries(CADPreview.Viewer PRIVATE CADPreview.Renderer)
  add_library(CADPreview.Shell SHARED src/shell.cpp src/shell.def)
  target_link_libraries(CADPreview.Shell PRIVATE CADPreview.Renderer uuid shlwapi)
  add_executable(CADPreview.NativeTests tests/native_tests.cpp)
  target_link_libraries(CADPreview.NativeTests PRIVATE CADPreview.Renderer uuid shlwapi)
  add_test(NAME native-shell-and-worker COMMAND CADPreview.NativeTests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)
  set_tests_properties(native-shell-and-worker PROPERTIES TIMEOUT 60)
  if(CADPREVIEW_WITH_OCCT)
    add_executable(CADPreview.StepFixtures tests/step_fixtures.cpp)
    target_include_directories(CADPreview.StepFixtures PRIVATE ${OpenCASCADE_INCLUDE_DIR})
    target_link_libraries(CADPreview.StepFixtures PRIVATE CADPreview.Core TKPrim TKBO TKBool)
  endif()
  if(MINGW)
    target_link_options(CADPreview.Worker PRIVATE -municode)
    target_link_options(CADPreview.Viewer PRIVATE -municode)
    set_target_properties(CADPreview.Shell PROPERTIES PREFIX "")
  endif()
  install(TARGETS CADPreview.Viewer CADPreview.Worker CADPreview.Shell RUNTIME DESTINATION . LIBRARY DESTINATION .)
endif()
